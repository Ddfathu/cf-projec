export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    const path = url.pathname;
    const userAgent = request.headers.get("user-agent") || "Unknown";

    if (!env.SUBDOMAIN_KV) {
      return jsonResponse({ 
        error: "KV Namespace belum di-bind! Buka Worker Settings -> Variables -> KV Namespace Bindings -> Set Variable Name jadi SUBDOMAIN_KV" 
      }, 500);
    }

    const MASTER_KEY = "dede123";

    if (path === "/api/admin/reset-pass" && request.method === "GET") {
      const key = url.searchParams.get("key");
      if (key !== MASTER_KEY) return jsonResponse({ error: "Master key salah!" }, 403);
      await env.SUBDOMAIN_KV.put("admin:secret", MASTER_KEY);
      return jsonResponse({ success: true, message: "Password admin berhasil direset jadi " + MASTER_KEY });
    }

    const isAuthorizedAdmin = async (secret) => {
      if (!secret) return false;
      const cleanSecret = String(secret).trim();
      if (cleanSecret === MASTER_KEY) return true;
      try {
        const storedSecret = await env.SUBDOMAIN_KV.get("admin:secret");
        return storedSecret && cleanSecret === storedSecret.trim();
      } catch (e) {
        return false;
      }
    };

    // Helper Kredensial Multi-Akun Cloudflare
    const getCfAccounts = async () => {
      return (await env.SUBDOMAIN_KV.get("admin:cf_accounts", { type: "json" })) || [];
    };

    const getCfAccountByEmail = async (email) => {
      const accounts = await getCfAccounts();
      return accounts.find(a => a.email.toLowerCase() === (email || "").toLowerCase().trim());
    };

    // --- API ENDPOINTS ADMIN (MULTI-AKUN CLOUDFLARE) ---

    if (path === "/api/admin/login" && request.method === "POST") {
      try {
        let adminSecret = "";
        try {
          const body = await request.json();
          adminSecret = body.adminSecret || "";
        } catch (_) {}

        const isValid = await isAuthorizedAdmin(adminSecret);
        if (!isValid) return jsonResponse({ error: "Password Admin Salah!" }, 401);

        const accounts = await getCfAccounts();
        return jsonResponse({ 
          success: true, 
          message: "Login Berhasil!",
          accountsCount: accounts.length
        });
      } catch (e) {
        return jsonResponse({ error: "Server Error: " + e.message }, 500);
      }
    }

    // Ambil list semua akun Cloudflare yang terdaftar di Admin
    if (path === "/api/admin/cf-accounts/list" && request.method === "POST") {
      try {
        const { adminSecret } = await request.json();
        if (!(await isAuthorizedAdmin(adminSecret))) return jsonResponse({ error: "Akses Ditolak!" }, 401);

        const accounts = await getCfAccounts();
        // Mask API Key agar aman ditampilkan
        const sanitized = accounts.map(a => ({
          email: a.email,
          createdAt: a.createdAt || "-"
        }));
        return jsonResponse({ success: true, accounts: sanitized });
      } catch (e) {
        return jsonResponse({ error: e.message }, 500);
      }
    }

    // Simpan/Tambah akun Cloudflare baru
    if (path === "/api/admin/cf-accounts/add" && request.method === "POST") {
      try {
        const { adminSecret, adminEmail, globalApiKey } = await request.json();
        if (!(await isAuthorizedAdmin(adminSecret))) return jsonResponse({ error: "Akses Ditolak!" }, 401);
        if (!adminEmail || !globalApiKey) return jsonResponse({ error: "Email & Global API Key wajib diisi!" }, 400);

        const cleanEmail = adminEmail.trim().toLowerCase();
        const cleanKey = globalApiKey.trim();

        // Validasi kredensial ke API Cloudflare sebelum simpan
        const cfRes = await fetch("https://api.cloudflare.com/client/v4/zones?per_page=1", {
          headers: {
            "X-Auth-Email": cleanEmail,
            "X-Auth-Key": cleanKey,
            "Content-Type": "application/json"
          }
        });
        const cfData = await cfRes.json();
        if (!cfData.success) {
          const msg = cfData.errors?.[0]?.message || "Kredensial Cloudflare tidak valid!";
          return jsonResponse({ error: msg }, 400);
        }

        let accounts = await getCfAccounts();
        accounts = accounts.filter(a => a.email.toLowerCase() !== cleanEmail);
        accounts.push({
          email: cleanEmail,
          apiKey: cleanKey,
          createdAt: new Date().toISOString()
        });

        await env.SUBDOMAIN_KV.put("admin:cf_accounts", JSON.stringify(accounts));
        return jsonResponse({ success: true, message: `Akun ${cleanEmail} berhasil ditambahkan!` });
      } catch (e) {
        return jsonResponse({ error: e.message }, 500);
      }
    }

    // Hapus satu akun Cloudflare
    if (path === "/api/admin/cf-accounts/delete" && request.method === "POST") {
      try {
        const { adminSecret, targetEmail } = await request.json();
        if (!(await isAuthorizedAdmin(adminSecret))) return jsonResponse({ error: "Akses Ditolak!" }, 401);

        let accounts = await getCfAccounts();
        accounts = accounts.filter(a => a.email.toLowerCase() !== (targetEmail || "").toLowerCase().trim());
        await env.SUBDOMAIN_KV.put("admin:cf_accounts", JSON.stringify(accounts));

        return jsonResponse({ success: true, message: `Akun ${targetEmail} berhasil dihapus!` });
      } catch (e) {
        return jsonResponse({ error: e.message }, 500);
      }
    }

    // Scan domain berdasarkan akun tertentu atau semua akun
    if (path === "/api/admin/fetch-zones" && request.method === "POST") {
      try {
        const { adminSecret, targetEmail } = await request.json();
        if (!(await isAuthorizedAdmin(adminSecret))) return jsonResponse({ error: "Password Admin Salah!" }, 401);

        const accounts = await getCfAccounts();
        if (!accounts || accounts.length === 0) {
          return jsonResponse({ error: "Belum ada akun Cloudflare yang terdaftar! Tambahkan akun terlebih dahulu." }, 400);
        }

        const accountsToScan = targetEmail 
          ? accounts.filter(a => a.email.toLowerCase() === targetEmail.toLowerCase().trim())
          : accounts;

        let allZones = [];

        for (const acc of accountsToScan) {
          try {
            const cfRes = await fetch("https://api.cloudflare.com/client/v4/zones?per_page=50", {
              headers: {
                "X-Auth-Email": acc.email,
                "X-Auth-Key": acc.apiKey,
                "Content-Type": "application/json"
              }
            });
            const cfData = await cfRes.json();
            if (cfData.success && cfData.result) {
              cfData.result.forEach(z => {
                allZones.push({
                  id: z.id,
                  name: z.name,
                  accountEmail: acc.email
                });
              });
            }
          } catch (_) {}
        }

        return jsonResponse({ success: true, zones: allZones });
      } catch (err) {
        return jsonResponse({ error: `Server Error: ${err.message}` }, 500);
      }
    }

    if (path === "/api/admin/set-ca" && request.method === "POST") {
      try {
        const { adminSecret, zoneId, ca, accountEmail } = await request.json();
        if (!(await isAuthorizedAdmin(adminSecret))) return jsonResponse({ error: "Unauthorized" }, 401);

        const acc = await getCfAccountByEmail(accountEmail);
        if (!acc) return jsonResponse({ error: "Akun Cloudflare pemilik zone ini tidak ditemukan!" }, 400);

        const caValue = ca === "google" ? "google" : "lets_encrypt";

        const cfRes = await fetch(`https://api.cloudflare.com/client/v4/zones/${zoneId}/ssl/universal/settings`, {
          method: "PATCH",
          headers: {
            "X-Auth-Email": acc.email,
            "X-Auth-Key": acc.apiKey,
            "Content-Type": "application/json"
          },
          body: JSON.stringify({ certificate_authority: caValue })
        });

        const cfData = await cfRes.json();
        if (!cfData.success) {
          return jsonResponse({ error: cfData.errors?.[0]?.message || "Gagal ubah CA!" }, 400);
        }

        const caName = caValue === "google" ? "Google Trust Services" : "Let's Encrypt";
        return jsonResponse({ success: true, message: `Berhasil mengubah Default CA ke ${caName}!` });
      } catch (err) {
        return jsonResponse({ error: "Server Error: " + err.message }, 500);
      }
    }

    if (path === "/api/admin/toggle-zone" && request.method === "POST") {
      try {
        const { adminSecret, zoneId, domainName, active, accountEmail } = await request.json();
        if (!(await isAuthorizedAdmin(adminSecret))) return jsonResponse({ error: "Unauthorized" }, 401);

        let activeZones = (await env.SUBDOMAIN_KV.get("active_zones", { type: "json" })) || {};
        if (active) {
          // Simpan zoneId beserta akun CF pemiliknya
          activeZones[domainName] = { zoneId, accountEmail };
        } else {
          delete activeZones[domainName];
        }

        await env.SUBDOMAIN_KV.put("active_zones", JSON.stringify(activeZones));
        return jsonResponse({ success: true, message: "Status domain diperbarui!" });
      } catch (e) {
        return jsonResponse({ error: e.message }, 500);
      }
    }

    if (path === "/api/admin/get-active-zones" && request.method === "GET") {
      const rawZones = (await env.SUBDOMAIN_KV.get("active_zones", { type: "json" })) || {};
      const activeZones = {};
      for (const [k, v] of Object.entries(rawZones)) {
        activeZones[k] = typeof v === "object" ? v.zoneId : v;
      }
      return jsonResponse({ activeZones, rawZones });
    }

    if (path === "/api/admin/users/list" && request.method === "POST") {
      try {
        const { adminSecret } = await request.json();
        if (!(await isAuthorizedAdmin(adminSecret))) return jsonResponse({ error: "Unauthorized" }, 401);

        const list = await env.SUBDOMAIN_KV.list({ prefix: "user:" });
        const users = list.keys.map(k => k.name.replace("user:", ""));

        return jsonResponse({ success: true, users });
      } catch (e) {
        return jsonResponse({ error: e.message }, 500);
      }
    }

    if (path === "/api/admin/users/detail" && request.method === "POST") {
      try {
        const { adminSecret, targetEmail } = await request.json();
        if (!(await isAuthorizedAdmin(adminSecret))) return jsonResponse({ error: "Unauthorized" }, 401);

        const userData = await env.SUBDOMAIN_KV.get(`user:${targetEmail}`, { type: "json" });
        if (!userData) return jsonResponse({ error: "User tidak ditemukan!" }, 404);

        const userDomains = (await env.SUBDOMAIN_KV.get(`user_subs:${targetEmail}`, { type: "json" })) || [];
        const customDomains = (await env.SUBDOMAIN_KV.get(`user_custom_zones:${targetEmail}`, { type: "json" })) || [];

        return jsonResponse({
          success: true,
          email: targetEmail,
          password: userData.password || "-",
          userAgent: userData.userAgent || "Belum tercatat",
          domains: userDomains,
          customDomains: customDomains
        });
      } catch (e) {
        return jsonResponse({ error: e.message }, 500);
      }
    }

    // --- API ENDPOINTS USER & DOMAIN PRIBADI ---

    if (path === "/api/register" && request.method === "POST") {
      const { email, password } = await request.json();
      if (!email || !password) return jsonResponse({ error: "Email & Password wajib!" }, 400);

      const existingUser = await env.SUBDOMAIN_KV.get(`user:${email}`);
      if (existingUser) return jsonResponse({ error: "Email sudah terdaftar!" }, 400);

      await env.SUBDOMAIN_KV.put(`user:${email}`, JSON.stringify({ password, userAgent }));
      return jsonResponse({ success: true, message: "Registrasi berhasil!" });
    }

    if (path === "/api/login" && request.method === "POST") {
      const { email, password } = await request.json();
      const userData = await env.SUBDOMAIN_KV.get(`user:${email}`, { type: "json" });

      if (!userData || userData.password !== password) {
        return jsonResponse({ error: "Email atau Password salah!" }, 401);
      }

      userData.userAgent = userAgent;
      await env.SUBDOMAIN_KV.put(`user:${email}`, JSON.stringify(userData));

      return jsonResponse({ success: true, email });
    }

    if (path === "/api/domains/public-list" && request.method === "POST") {
      const { email } = await request.json().catch(() => ({}));
      const activeZones = (await env.SUBDOMAIN_KV.get("active_zones", { type: "json" })) || {};
      const customZones = (await env.SUBDOMAIN_KV.get(`user_custom_zones:${email}`, { type: "json" })) || [];

      const publicDomains = Object.keys(activeZones);
      const userOwnedDomains = customZones.map(z => z.domainName);

      const combined = Array.from(new Set([...publicDomains, ...userOwnedDomains]));
      return jsonResponse({ domains: combined });
    }

    if (path === "/api/user/custom-zone/add" && request.method === "POST") {
      try {
        const { email, domainName } = await request.json();
        if (!email || !domainName) return jsonResponse({ error: "Nama domain wajib diisi!" }, 400);

        const cleanDomain = domainName.trim().toLowerCase().replace(/^https?:\/\//, '').replace(/\/.*$/, '');
        const accounts = await getCfAccounts();
        if (!accounts || accounts.length === 0) {
          return jsonResponse({ error: "Admin belum mengatur kredensial Cloudflare!" }, 400);
        }

        // Gunakan akun pertama sebagai default pendaftaran zone baru
        const defaultAcc = accounts[0];

        let zoneRes = await fetch(`https://api.cloudflare.com/client/v4/zones?name=${cleanDomain}`, {
          headers: {
            "X-Auth-Email": defaultAcc.email,
            "X-Auth-Key": defaultAcc.apiKey,
            "Content-Type": "application/json"
          }
        });
        let zoneData = await zoneRes.json();
        let targetZone = zoneData.result?.[0];

        if (!targetZone) {
          const createRes = await fetch("https://api.cloudflare.com/client/v4/zones", {
            method: "POST",
            headers: {
              "X-Auth-Email": defaultAcc.email,
              "X-Auth-Key": defaultAcc.apiKey,
              "Content-Type": "application/json"
            },
            body: JSON.stringify({ name: cleanDomain, jump_start: true })
          });
          const createData = await createRes.json();
          if (!createData.success) {
            return jsonResponse({ error: createData.errors?.[0]?.message || "Gagal mendaftarkan domain ke Cloudflare!" }, 400);
          }
          targetZone = createData.result;
        }

        const nameServers = targetZone.name_servers || ["ns1.cloudflare.com", "ns2.cloudflare.com"];
        const zoneId = targetZone.id;

        let userZones = (await env.SUBDOMAIN_KV.get(`user_custom_zones:${email}`, { type: "json" })) || [];
        if (!userZones.some(z => z.domainName === cleanDomain)) {
          userZones.push({
            domainName: cleanDomain,
            zoneId: zoneId,
            nameServers: nameServers,
            accountEmail: defaultAcc.email,
            createdAt: new Date().toISOString()
          });
          await env.SUBDOMAIN_KV.put(`user_custom_zones:${email}`, JSON.stringify(userZones));
        }

        return jsonResponse({
          success: true,
          message: "Domain pribadi berhasil ditambahkan!",
          domainName: cleanDomain,
          zoneId: zoneId,
          nameServers: nameServers
        });
      } catch (err) {
        return jsonResponse({ error: "Server Error: " + err.message }, 500);
      }
    }

    if (path === "/api/user/custom-zone/list" && request.method === "POST") {
      try {
        const { email } = await request.json();
        const userZones = (await env.SUBDOMAIN_KV.get(`user_custom_zones:${email}`, { type: "json" })) || [];
        const activeZones = (await env.SUBDOMAIN_KV.get("active_zones", { type: "json" })) || {};

        const result = userZones.map(z => ({
          ...z,
          isPublic: !!activeZones[z.domainName]
        }));

        return jsonResponse({ success: true, customZones: result });
      } catch (err) {
        return jsonResponse({ error: err.message }, 500);
      }
    }

    if (path === "/api/user/custom-zone/toggle-public" && request.method === "POST") {
      try {
        const { email, domainName, isPublic } = await request.json();
        const userZones = (await env.SUBDOMAIN_KV.get(`user_custom_zones:${email}`, { type: "json" })) || [];
        const target = userZones.find(z => z.domainName === domainName);

        if (!target) return jsonResponse({ error: "Domain ini bukan milik Anda!" }, 403);

        let activeZones = (await env.SUBDOMAIN_KV.get("active_zones", { type: "json" })) || {};
        if (isPublic) {
          activeZones[domainName] = { zoneId: target.zoneId, accountEmail: target.accountEmail || "" };
        } else {
          delete activeZones[domainName];
        }

        await env.SUBDOMAIN_KV.put("active_zones", JSON.stringify(activeZones));
        return jsonResponse({ success: true, message: `Status publik domain ${domainName} diperbarui!` });
      } catch (err) {
        return jsonResponse({ error: err.message }, 500);
      }
    }

    if (path === "/api/user/custom-zone/delete" && request.method === "POST") {
      try {
        const { email, domainName } = await request.json();
        let userZones = (await env.SUBDOMAIN_KV.get(`user_custom_zones:${email}`, { type: "json" })) || [];
        userZones = userZones.filter(z => z.domainName !== domainName);
        await env.SUBDOMAIN_KV.put(`user_custom_zones:${email}`, JSON.stringify(userZones));

        let activeZones = (await env.SUBDOMAIN_KV.get("active_zones", { type: "json" })) || {};
        delete activeZones[domainName];
        await env.SUBDOMAIN_KV.put("active_zones", JSON.stringify(activeZones));

        return jsonResponse({ success: true, message: `Domain ${domainName} berhasil dihapus!` });
      } catch (err) {
        return jsonResponse({ error: err.message }, 500);
      }
    }

    if (path === "/api/domains/add" && request.method === "POST") {
      try {
        const body = await request.json();
        const prefix = body.prefix || body.subdomain;
        let parentDomain = body.parentDomain;

        if (!prefix || !prefix.trim()) return jsonResponse({ error: "Nama subdomain wajib diisi!" }, 400);
        if (!parentDomain || !parentDomain.trim()) return jsonResponse({ error: "Pilih domain penyedia!" }, 400);

        while (parentDomain.startsWith('.')) {
          parentDomain = parentDomain.substring(1);
        }

        const cleanPrefix = prefix.trim().toLowerCase();
        const cleanParent = parentDomain.trim().toLowerCase();
        const fullSubdomain = `${cleanPrefix}.${cleanParent}`;

        const existingSub = await env.SUBDOMAIN_KV.get(`subdomain:${fullSubdomain}`);
        if (existingSub) return jsonResponse({ error: "Subdomain ini sudah ada yang punya!" }, 400);

        await env.SUBDOMAIN_KV.put(`subdomain:${fullSubdomain}`, JSON.stringify({ owner: body.email }));

        let userSubs = (await env.SUBDOMAIN_KV.get(`user_subs:${body.email}`, { type: "json" })) || [];
        userSubs.push(fullSubdomain);
        await env.SUBDOMAIN_KV.put(`user_subs:${body.email}`, JSON.stringify(userSubs));

        return jsonResponse({ success: true, message: `Berhasil membuat ${fullSubdomain}` });
      } catch (err) {
        return jsonResponse({ error: "Server Error: " + err.message }, 500);
      }
    }

    if (path === "/api/domains/list" && request.method === "POST") {
      const { email } = await request.json();
      const userSubs = (await env.SUBDOMAIN_KV.get(`user_subs:${email}`, { type: "json" })) || [];
      return jsonResponse({ subdomains: userSubs });
    }

    if (path === "/api/domains/delete" && request.method === "POST") {
      try {
        const { email, fullSubdomain } = await request.json();
        if (!fullSubdomain) return jsonResponse({ error: "Subdomain tidak valid!" }, 400);

        await env.SUBDOMAIN_KV.delete(`subdomain:${fullSubdomain}`);

        let userSubs = (await env.SUBDOMAIN_KV.get(`user_subs:${email}`, { type: "json" })) || [];
        userSubs = userSubs.filter(sub => sub !== fullSubdomain);
        await env.SUBDOMAIN_KV.put(`user_subs:${email}`, JSON.stringify(userSubs));

        return jsonResponse({ success: true, message: `Subdomain ${fullSubdomain} telah dihapus permanen!` });
      } catch (err) {
        return jsonResponse({ error: "Server Error: " + err.message }, 500);
      }
    }

    // --- API DNS OPERATIONS (MENDUKUNG MULTI-AKUN) ---

    const resolveZoneInfo = async (fullSubdomain) => {
      const rawZones = (await env.SUBDOMAIN_KV.get("active_zones", { type: "json" })) || {};
      let parentDomain = Object.keys(rawZones).find(d => fullSubdomain === d || fullSubdomain.endsWith("." + d));
      
      let zoneId = null;
      let accountEmail = null;

      if (parentDomain) {
        const val = rawZones[parentDomain];
        if (typeof val === "object") {
          zoneId = val.zoneId;
          accountEmail = val.accountEmail;
        } else {
          zoneId = val;
        }
      } else {
        const userList = await env.SUBDOMAIN_KV.list({ prefix: "user_custom_zones:" });
        for (const k of userList.keys) {
          const zones = (await env.SUBDOMAIN_KV.get(k.name, { type: "json" })) || [];
          const match = zones.find(z => fullSubdomain === z.domainName || fullSubdomain.endsWith("." + z.domainName));
          if (match) {
            zoneId = match.zoneId;
            parentDomain = match.domainName;
            accountEmail = match.accountEmail;
            break;
          }
        }
      }

      if (!zoneId) return null;

      const accounts = await getCfAccounts();
      let matchedAcc = null;
      if (accountEmail) {
        matchedAcc = accounts.find(a => a.email.toLowerCase() === accountEmail.toLowerCase());
      }
      if (!matchedAcc && accounts.length > 0) {
        matchedAcc = accounts[0];
      }

      if (!matchedAcc) return null;

      return {
        zoneId,
        parentDomain,
        adminEmail: matchedAcc.email,
        globalApiKey: matchedAcc.apiKey
      };
    };

    const formatRecordName = (name, fullSubdomain, parentDomain, type) => {
      let rawName = (name || "").trim().toLowerCase();
      const cleanSub = fullSubdomain.trim().toLowerCase();
      const cleanParent = (parentDomain || "").trim().toLowerCase();

      let userSubPrefix = "";
      if (cleanParent && cleanSub.endsWith("." + cleanParent)) {
        userSubPrefix = cleanSub.replace("." + cleanParent, "");
      }

      if (rawName === "@" || rawName === cleanSub || (userSubPrefix && rawName === userSubPrefix)) {
        rawName = "";
      } else if (rawName.endsWith("." + cleanSub)) {
        rawName = rawName.replace("." + cleanSub, "");
      } else if (userSubPrefix && rawName.endsWith("." + userSubPrefix)) {
        rawName = rawName.slice(0, -(userSubPrefix.length + 1));
      }

      if (String(type).toUpperCase() === "TXT") {
        const baseTarget = cleanParent || cleanSub;
        return rawName ? `${rawName}.${baseTarget}` : baseTarget;
      }

      return rawName ? `${rawName}.${cleanSub}` : cleanSub;
    };

    if (path === "/api/dns/list" && request.method === "POST") {
      try {
        const { fullSubdomain } = await request.json();
        const zoneInfo = await resolveZoneInfo(fullSubdomain);

        if (!zoneInfo) return jsonResponse({ error: "Zone atau Kredensial Akun Cloudflare tidak ditemukan!" }, 400);

        const cfRes = await fetch(`https://api.cloudflare.com/client/v4/zones/${zoneInfo.zoneId}/dns_records?per_page=100`, {
          headers: {
            "X-Auth-Email": zoneInfo.adminEmail,
            "X-Auth-Key": zoneInfo.globalApiKey,
            "Content-Type": "application/json"
          }
        });

        const cfData = await cfRes.json();
        if (!cfData.success) return jsonResponse({ error: "Gagal mengambil data DNS dari Cloudflare" }, 400);

        const records = cfData.result.filter(r => {
          if (r.name === fullSubdomain || r.name.endsWith(`.${fullSubdomain}`)) return true;
          if (r.type === "TXT" && zoneInfo.parentDomain) {
            return r.name === zoneInfo.parentDomain || r.name.endsWith(`.${zoneInfo.parentDomain}`);
          }
          return false;
        });

        return jsonResponse({ success: true, records, parentDomain: zoneInfo.parentDomain });
      } catch (err) {
        return jsonResponse({ error: err.message }, 500);
      }
    }

    if (path === "/api/dns/add" && request.method === "POST") {
      const { fullSubdomain, type, name, content, ttl, proxied } = await request.json();
      const zoneInfo = await resolveZoneInfo(fullSubdomain);

      if (!zoneInfo) return jsonResponse({ error: "Kredensial akun domain utama belum diatur!" }, 500);

      const recordName = formatRecordName(name, fullSubdomain, zoneInfo.parentDomain, type);
      const isProxyable = ["A", "AAAA", "CNAME"].includes(String(type).toUpperCase());
      const finalProxied = isProxyable ? Boolean(proxied) : false;

      const cfRes = await fetch(`https://api.cloudflare.com/client/v4/zones/${zoneInfo.zoneId}/dns_records`, {
        method: "POST",
        headers: {
          "X-Auth-Email": zoneInfo.adminEmail,
          "X-Auth-Key": zoneInfo.globalApiKey,
          "Content-Type": "application/json"
        },
        body: JSON.stringify({ 
          type, 
          name: recordName, 
          content, 
          ttl: finalProxied ? 1 : (parseInt(ttl) || 1), 
          proxied: finalProxied 
        })
      });

      const cfData = await cfRes.json();
      if (!cfData.success) return jsonResponse({ error: cfData.errors?.[0]?.message || "Gagal inject DNS Record!" }, 400);

      return jsonResponse({ success: true, record: cfData.result });
    }

    if (path === "/api/dns/edit" && request.method === "POST") {
      try {
        const { fullSubdomain, recordId, type, name, content, ttl, proxied } = await request.json();
        const zoneInfo = await resolveZoneInfo(fullSubdomain);

        if (!zoneInfo) return jsonResponse({ error: "Zone tidak ditemukan!" }, 400);

        const recordName = formatRecordName(name, fullSubdomain, zoneInfo.parentDomain, type);
        const isProxyable = ["A", "AAAA", "CNAME"].includes(String(type).toUpperCase());
        const finalProxied = isProxyable ? Boolean(proxied) : false;

        const cfRes = await fetch(`https://api.cloudflare.com/client/v4/zones/${zoneInfo.zoneId}/dns_records/${recordId}`, {
          method: "PUT",
          headers: {
            "X-Auth-Email": zoneInfo.adminEmail,
            "X-Auth-Key": zoneInfo.globalApiKey,
            "Content-Type": "application/json"
          },
          body: JSON.stringify({ 
            type, 
            name: recordName, 
            content, 
            ttl: finalProxied ? 1 : (parseInt(ttl) || 1), 
            proxied: finalProxied 
          })
        });

        const cfData = await cfRes.json();
        if (!cfData.success) return jsonResponse({ error: cfData.errors?.[0]?.message || "Gagal update record!" }, 400);

        return jsonResponse({ success: true, message: "Record berhasil diperbarui!" });
      } catch (err) {
        return jsonResponse({ error: err.message }, 500);
      }
    }

    if (path === "/api/dns/set-ns" && request.method === "POST") {
      const { fullSubdomain, ns1, ns2 } = await request.json();
      const zoneInfo = await resolveZoneInfo(fullSubdomain);

      if (!zoneInfo) return jsonResponse({ error: "Domain utama tidak valid!" }, 500);

      const addNS = async (nsValue) => {
        return fetch(`https://api.cloudflare.com/client/v4/zones/${zoneInfo.zoneId}/dns_records`, {
          method: "POST",
          headers: { "X-Auth-Email": zoneInfo.adminEmail, "X-Auth-Key": zoneInfo.globalApiKey, "Content-Type": "application/json" },
          body: JSON.stringify({ type: "NS", name: fullSubdomain, content: nsValue, ttl: 1 })
        });
      };

      if (ns1) await addNS(ns1);
      if (ns2) await addNS(ns2);

      return jsonResponse({ success: true, message: "NS Record berhasil diterapkan!" });
    }

    if (path === "/api/dns/delete-record" && request.method === "POST") {
      try {
        const { fullSubdomain, recordId } = await request.json();
        const zoneInfo = await resolveZoneInfo(fullSubdomain);

        if (!zoneInfo) return jsonResponse({ error: "Zone tidak ditemukan!" }, 400);

        const cfRes = await fetch(`https://api.cloudflare.com/client/v4/zones/${zoneInfo.zoneId}/dns_records/${recordId}`, {
          method: "DELETE",
          headers: { "X-Auth-Email": zoneInfo.adminEmail, "X-Auth-Key": zoneInfo.globalApiKey, "Content-Type": "application/json" }
        });

        const cfData = await cfRes.json();
        if (!cfData.success) return jsonResponse({ error: "Gagal menghapus record!" }, 400);

        return jsonResponse({ success: true, message: "Record berhasil dihapus!" });
      } catch (err) {
        return jsonResponse({ error: err.message }, 500);
      }
    }

    if (path === "/admin") return new Response(getAdminUI(), { headers: { "Content-Type": "text/html;charset=UTF-8" } });
    return new Response(getUserUI(), { headers: { "Content-Type": "text/html;charset=UTF-8" } });
  }
};

function jsonResponse(data, status = 200) {
  return new Response(JSON.stringify(data), { status, headers: { "Content-Type": "application/json" } });
}

    async function addCustomDomain() {
      const input = document.getElementById("custom-domain-name");
      const domainName = input.value.trim();
      if (!domainName) return alert("Masukkan nama domain pribadimu!");

      const btn = event.target;
      btn.innerText = "Mendaftarkan...";
      btn.disabled = true;

      try {
        const res = await fetch("/api/user/custom-zone/add", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ email: currentUser, domainName })
        });
        const data = await res.json();

        if (data.success) {
          alert("Domain berhasil ditambahkan! Silakan ubah Nameserver domain kamu ke:\\n" + data.nameServers.join("\\n"));

          loadCustomDomains();
          loadPublicDomains();

        }
      } catch (e) {
        alert("Error: " + e.message);
      } finally {
        btn.innerText = "Daftarkan Domain";
        btn.disabled = false;
      }

    async function loadDnsOnly() {
      const dnsListDiv = document.getElementById("dns-records-list");
      dnsListDiv.innerHTML = '<p class="text-xs text-slate-400">Loading DNS...</p>';

      try {
        const res = await fetch("/api/dns/list", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ fullSubdomain: activeSubdomainContext })
        });
        const data = await res.json();

        if(!data.success) {
          dnsListDiv.innerHTML = '<p class="text-xs text-red-500">' + data.error + '</p>';
          return;
        }

        activeParentDomain = data.parentDomain || "";
        const dnsRecords = data.records.filter(r => r.type !== 'NS');
        dnsListDiv.innerHTML = "";

        if(dnsRecords.length === 0) {
          dnsListDiv.innerHTML = '<p class="text-xs text-slate-400 italic">Belum ada DNS Record terpasang.</p>';
        } else {
          dnsRecords.forEach(function(r) {
            var rBox = document.createElement("div");
            rBox.className = "flex justify-between items-center bg-slate-50 p-2.5 rounded-xl border border-slate-200 text-xs gap-1 hover:border-slate-300 transition";
            
            var cloudBadge = "";
            var isProxyable = ["A", "AAAA", "CNAME"].includes(r.type);
            if (isProxyable) {
              if (r.proxied) {
                cloudBadge = '<span title="Proxied (Klik untuk jadikan DNS Only)" class="cursor-pointer inline-flex items-center text-orange-500 hover:text-orange-600 bg-orange-100 px-1.5 py-0.5 rounded-md font-bold text-[10px]">' +
                  '<svg class="w-3.5 h-3.5 mr-0.5 fill-current" viewBox="0 0 24 24"><path d="M19.35 10.04C18.67 6.59 15.64 4 12 4 9.11 4 6.6 5.64 5.35 8.04 2.34 8.36 0 10.91 0 14c0 3.31 2.69 6 6 6h13c2.76 0 5-2.24 5-5 0-2.64-2.05-4.78-4.65-4.96z"/></svg>Proxied</span>';
              } else {
                cloudBadge = '<span title="DNS Only (Klik untuk aktifkan Proxied)" class="cursor-pointer inline-flex items-center text-slate-500 hover:text-slate-700 bg-slate-200 px-1.5 py-0.5 rounded-md font-bold text-[10px]">' +
                  '<svg class="w-3.5 h-3.5 mr-0.5 fill-current" viewBox="0 0 24 24"><path d="M19.35 10.04C18.67 6.59 15.64 4 12 4 9.11 4 6.6 5.64 5.35 8.04 2.34 8.36 0 10.91 0 14c0 3.31 2.69 6 6 6h13c2.76 0 5-2.24 5-5 0-2.64-2.05-4.78-4.65-4.96z"/></svg>DNS</span>';
              }
            }

            rBox.innerHTML = '<div class="truncate mr-1 flex items-center gap-1.5">' +
                '<span class="font-bold text-blue-600 bg-blue-100 px-1.5 py-0.5 rounded text-[10px]">' + r.type + '</span>' +
                '<div class="btn-toggle-quick-proxy">' + cloudBadge + '</div>' +
                '<span class="text-slate-700 font-medium truncate">' + r.name + '</span>' +
                '<span class="text-slate-400">&rarr;</span>' +
                '<span class="text-slate-500 font-mono truncate max-w-[120px]">' + r.content + '</span>' +
              '</div>' +
              '<div class="flex gap-1 shrink-0">' +
                '<button class="btn-edt bg-amber-50 text-amber-600 border border-amber-200 text-[10px] px-2 py-1 rounded font-bold hover:bg-amber-100 transition">Edit</button>' +
                '<button class="btn-del bg-red-50 text-red-600 border border-red-200 text-[10px] px-2 py-1 rounded font-bold hover:bg-red-100 transition">Hapus</button>' +
              '</div>';

            var qProxy = rBox.querySelector(".btn-toggle-quick-proxy span");
            if (qProxy) {
              qProxy.onclick = function() {
                toggleQuickProxy(r.id, r.type, r.name, r.content, r.ttl, !r.proxied);
              };
            }

            rBox.querySelector(".btn-edt").onclick = function() { fillDnsFormForEdit(r.id, r.type, r.name, r.content, r.ttl, r.proxied); };
            rBox.querySelector(".btn-del").onclick = function() { deleteDnsRecord(r.id, 'dns'); };

            dnsListDiv.appendChild(rBox);
          });
        }
      } catch(e) {
        dnsListDiv.innerHTML = '<p class="text-xs text-red-500">Gagal memuat data DNS.</p>';
      }
    }

    async function toggleQuickProxy(id, type, name, content, ttl, newProxied) {
      const res = await fetch("/api/dns/edit", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          fullSubdomain: activeSubdomainContext,
          recordId: id,
          type: type,
          name: name,
          content: content,
          ttl: newProxied ? 1 : ttl,
          proxied: newProxied
        })
      });
      const data = await res.json();
      if (data.success) {
        loadDnsOnly();
      } else {
        alert(data.error || "Gagal mengubah proxy");
      }
    }

    function fillDnsFormForEdit(id, type, name, content, ttl, proxied) {
      document.getElementById("editing-record-id").value = id;
      document.getElementById("dns-type").value = type;
      handleTypeChange();
      
      let hostPrefix = name;
      if (type === "TXT" && activeParentDomain) {
        if (name === activeParentDomain) hostPrefix = "@";
        else if (name.endsWith("." + activeParentDomain)) hostPrefix = name.replace("." + activeParentDomain, "");
      } else {
        if (name === activeSubdomainContext) hostPrefix = "@";
        else if (name.endsWith("." + activeSubdomainContext)) hostPrefix = name.replace("." + activeSubdomainContext, "");
      }
      
      document.getElementById("dns-name").value = hostPrefix;
      document.getElementById("dns-content").value = content;
      document.getElementById("dns-ttl").value = ttl;

      setProxyState(Boolean(proxied));

      document.getElementById("btn-save-dns").innerText = "Update DNS Record";
      document.getElementById("btn-save-dns").className = "bg-amber-600 hover:bg-amber-700 text-white flex-1 py-2 rounded-lg font-bold text-xs transition shadow-sm";
      document.getElementById("btn-cancel-dns").classList.remove("hidden");
    }

    function resetDnsForm() {
      document.getElementById("editing-record-id").value = "";
      document.getElementById("dns-name").value = "";
      document.getElementById("dns-content").value = "";
      document.getElementById("dns-ttl").value = "1";
      document.getElementById("dns-type").value = "A";
      handleTypeChange();
      setProxyState(false);

      document.getElementById("btn-save-dns").innerText = "Simpan DNS";
      document.getElementById("btn-save-dns").className = "bg-blue-600 hover:bg-blue-700 text-white flex-1 py-2 rounded-lg font-bold text-xs transition shadow-sm";
      document.getElementById("btn-cancel-dns").classList.add("hidden");
    }

    async function saveDnsRecord() {
      const recordId = document.getElementById("editing-record-id").value;
      const type = document.getElementById("dns-type").value;
      const name = document.getElementById("dns-name").value;
      const content = document.getElementById("dns-content").value;
      const ttl = document.getElementById("dns-ttl").value;

      if(!content) return alert("Isi / Target record wajib diisi!");

      const endpoint = recordId ? "/api/dns/edit" : "/api/dns/add";
      const bodyPayload = { 
        fullSubdomain: activeSubdomainContext, 
        recordId, 
        type, 
        name, 
        content, 
        ttl,
        proxied: isProxiedActive 
      };

      const res = await fetch(endpoint, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(bodyPayload) });
      const data = await res.json();
      alert(data.message || (data.success ? "DNS Record Berhasil Disimpan!" : data.error));
      
      if (data.success) {
        resetDnsForm();
        loadDnsOnly();
      }
    }

    function openNsModal(fullSubdomain) {
      activeSubdomainContext = fullSubdomain;
      document.getElementById("modal-ns-sub").innerText = fullSubdomain;
      document.getElementById("ns-modal").classList.remove("hidden");
      loadNsOnly();
    }

    function closeNsModal() {
      document.getElementById("ns-modal").classList.add("hidden");
    }

    async function loadNsOnly() {
      const nsListDiv = document.getElementById("ns-records-list");
      nsListDiv.innerHTML = '<p class="text-xs text-slate-400">Loading NS...</p>';

      try {
        const res = await fetch("/api/dns/list", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ fullSubdomain: activeSubdomainContext })
        });
        const data = await res.json();

        if(!data.success) {
          nsListDiv.innerHTML = '<p class="text-xs text-red-500">' + data.error + '</p>';
          return;
        }

        const nsRecords = data.records.filter(r => r.type === 'NS');
        nsListDiv.innerHTML = "";

        if(nsRecords.length === 0) {
          nsListDiv.innerHTML = '<p class="text-xs text-slate-400 italic">Belum ada Custom NS terpasang.</p>';
        } else {
          nsRecords.forEach(function(r, idx) {
            var nsBox = document.createElement("div");
            nsBox.className = "flex justify-between items-center bg-slate-50 p-2 rounded-lg border border-slate-200 text-xs gap-1";
            
            nsBox.innerHTML = '<div class="truncate mr-1">' +
                '<span class="font-bold text-purple-600 bg-purple-100 px-1.5 py-0.5 rounded text-[10px] mr-1">NS ' + (idx + 1) + '</span>' +
                '<span class="text-slate-700 font-mono">' + r.content + '</span>' +
              '</div>' +
              '<button class="btn-del bg-red-50 text-red-600 border border-red-200 text-[10px] px-2 py-1 rounded font-bold hover:bg-red-100 transition shrink-0">Hapus</button>';

            nsBox.querySelector(".btn-del").onclick = function() { deleteDnsRecord(r.id, 'ns'); };

            nsListDiv.appendChild(nsBox);
          });
        }
      } catch(e) {
        nsListDiv.innerHTML = '<p class="text-xs text-red-500">Gagal memuat data NS.</p>';
      }
    }

    async function deleteDnsRecord(recordId, source) {
      if(!confirm("Hapus record ini secara permanen dari Cloudflare?")) return;

      const res = await fetch("/api/dns/delete-record", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ fullSubdomain: activeSubdomainContext, recordId })
      });
      const data = await res.json();
      alert(data.message || data.error);
      if(data.success) {
        if(source === 'dns') loadDnsOnly();
        else loadNsOnly();
      }
    }

    function copyDomain(text) {
      navigator.clipboard.writeText(text).then(function() {
        alert("Domain disalin: " + text);
      }).catch(function() {
        alert("Gagal menyalin domain!");
      });
    }

    async function deleteDomain(fullSubdomain) {
      if (!confirm("Yakin ingin menghapus " + fullSubdomain + " secara permanen?")) return;

      const res = await fetch("/api/domains/delete", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ email: currentUser, fullSubdomain })
      });
      const data = await res.json();
      alert(data.message || data.error);
      if (data.success) loadSubdomains();
    }

    async function setNS() {
      const ns1 = document.getElementById("ns1").value;
      const ns2 = document.getElementById("ns2").value;

      if(!ns1 && !ns2) return alert("Isi minimal satu Nameserver!");

      const res = await fetch("/api/dns/set-ns", { 
        method: "POST", 
        headers: { "Content-Type": "application/json" }, 
        body: JSON.stringify({ fullSubdomain: activeSubdomainContext, ns1, ns2 }) 
      });
      const data = await res.json();
      alert(data.message || data.error);
      if (data.success) {
        document.getElementById("ns1").value = "";
        document.getElementById("ns2").value = "";
        loadNsOnly();
      }
    }
  </script>
</body>
</html>`;
